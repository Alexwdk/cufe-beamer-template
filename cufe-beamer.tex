\section{研究方法与结果}
\cufechapter{02}{研究方法与结果}
\begin{frame}{模型设定与变量说明}
  \begin{block}{示例：面板数据回归模型}
    \[
      y_{it}=\alpha+\beta x_{it}+\boldsymbol{\gamma}^{\mathsf T}
      \mathbf{Z}_{it}+\mu_i+\lambda_t+\varepsilon_{it}
    \]
  \end{block}
  \begin{itemize}
    \item $y_{it}$：被解释变量；$x_{it}$：核心解释变量。
    \item $\mathbf{Z}_{it}$：控制变量；$\mu_i$、$\lambda_t$：个体与时间固定效应。
    \item $\beta$：待估计的核心参数；识别解释需结合研究设计。
  \end{itemize}
\end{frame}

\begin{frame}{主要结果展示}
  \framesubtitle{以下为排版示例，所有数值均为演示数据}
  \centering
  \begin{tabular}{lccc}

    个体固定效应 & 否 & 否 & 是 \\
    样本量 & 1,200 & 1,200 & 1,200 \\
    \bottomrule
  \end{tabular}
  \vspace{4mm}
  \begin{block}{结果解读}
    在此解释估计结果、经济含义以及结论成立的条件。
  \end{block}
\end{frame}
